% Original presentation frame 39
\begin{frame}[t,shrink=5]{Normalized mass becomes physical density only after coordinate scales define the bin areas}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{block}{Shape representation}
\[\sum_{a,b}H_{ij}^{(s)}[a,b]=1.\]The CVAE learns projected shape in normalized probability-mass coordinates.
\end{block}

\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Physical reconstruction}
\[\rho_{ij}^{(s)}[a,b]=\frac{H_{ij}^{(s)}[a,b]}{\Delta\xi_{i,a}^{(s)}\Delta\xi_{j,b}^{(s)}}.\]Bin edges are determined by the predicted centroids and RMS spreads.
\end{block}

\end{column}
\end{columns}
\begin{block}{Terminology}
RMS spread remains defined for non-Gaussian distributions with finite second moment but does not characterize tails, asymmetry or multimodality. Only the transverse spatial spreads are conventional beam sizes under the stated optics assumptions.
\end{block}

\end{frame}

% Original presentation frame 40
\begin{frame}[t,shrink=5]{The CVAE and endpoint propagators use different objectives}
\begin{block}{Snapshot CVAE pretraining}
\[\mathcal L_{\rm CVAE}=\mathcal L_{\rm rec}+10^{-2}\mathcal L_{\rm KL}+10^{-3}\mathcal L_\sigma+10^{-3}\mathcal L_{\rm cent}.\]
\end{block}
\begin{block}{Endpoint propagation}
\[\mathcal L_{\rm prop}=\mathcal L_{{\rm hist},N}+\beta_{\rm KL}\mathcal L_{\rm KL}+10\mathcal L_{\log\sigma,N}+10\mathcal L_{m,N}.\]
\end{block}
\begin{block}{Training scope}
The decoder is frozen during the window study, encoder/posterior heads, tokenizer and latent propagator remain trainable. Later, the distribution branch is frozen while the physical-statistics head is fine-tuned. No intermediate physical checkpoint appears in the loss.
\end{block}

\sources{\parencite{kingma2014vae,sohn2015cvae}.}

\end{frame}
